მთავარი

რა ხდება for-ის შიგნით

for x in [10, 20, 30]:
    print(x)

while True:
    try:

ორი ცნება Iterable (გასავლელი) — ობიექტი, რომელზეც for-ით გავლა შეიძლება: list, str, dict, ფაილი… Iterator (მიმდევრობის მკითხველი) — ობიექტი, რომელსაც ახსოვს, სად გაჩერდა, და next()-ზე შემდეგს აბრუნებს
ეს ლექციის გასაღები სლაიდია. სტუდენტებს აქამდე for „ჯადოსნური“ ეგონათ — აქ ხდება ცხადი, რომ ისიც ჩვეულებრივი მექანიზმია.

შექმნა

საკუთარი იტერატორი

class Countdown:
    """უკუთვლა n-დან 1-მდე."""

    def __init__(self, start):
        self.current = start

    def __iter__(self):
        return self                # თავად არის იტერატორი

    def __next__(self):
        if self.current <= 0:
            raise StopIteration    # სიგნალი: დასრულდა
        value = self.current
        self.current -= 1
        return value


for n in Countdown(5):
    print(n, end=" ")              # 5 4 3 2 1
ორი მეთოდი — ესაა მთელი „კონტრაქტი“__iter__() აბრუნებს იტერატორს, __next__() — შემდეგ ელემენტს ან ისვრის StopIteration-ს. ვინც ამ ორ მეთოდს ატარებს, მასზე for იმუშავებს.

კლასები კურსის პროგრამაში არაა — აქ ის მხოლოდ მექანიზმის საჩვენებლადაა. ქვემოთ ბევრად მარტივი გზაა.

მარტივი გზა

გენერატორი — yield

class Countdown:
    def __init__(self, start):
        self.current = start
    def __iter__(self):
        return self
    def __next__(self):
        if self.current <= 0:
            raise StopIteration
        v = self.current
        self.current -= 1
        return v
def countdown(start):
    while start > 0:
        yield start
        start -= 1


for n in countdown(5):
    print(n, end=" ")     # 5 4 3 2 1
yield vs returnreturn ფუნქციას ამთავრებს. yield მნიშვნელობას აბრუნებს და ფუნქციას „აპაუზებს“ — შემდეგი next() სწორედ იმ ადგილიდან გააგრძელებს.
ეს დემონსტრაცია დებაგერით შესანიშნავად მუშაობს: breakpoint yield-ზე და ნახავენ, როგორ „ბრუნდება“ შესრულება იმავე ხაზზე.

უპირატესობა

„ზარმაცი“ გამოთვლა (lazy evaluation)

import sys

# სია — ყველა ელემენტი მაშინვე მეხსიერებაშია
squares_list = [n ** 2 for n in range(1_000_000)]
print(sys.getsizeof(squares_list))     # ~8 400 000 ბაიტი

# გენერატორი — ელემენტი მხოლოდ საჭიროებისას იქმნება

print(sys.getsizeof(squares_gen))      # ~200 ბაიტი

print(sum(squares_gen))                # მუშაობს, მეხსიერების გარეშე

პრაქტიკა

გენერატორები რეალურ ამოცანაში

def read_lines(path):
    """ფაილს ხაზ-ხაზ კითხულობს, ცარიელებს ტოვებს."""
    with open(path, encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if line:
                yield line


def only_errors(lines):
    """მხოლოდ შეცდომების ხაზები."""
    for line in lines:
        if "ERROR" in line:
            yield line


def first_words(lines):
    for line in lines:
        yield line.split()[0]


# ჯაჭვი — არცერთი შუალედური სია არ იქმნება
for word in first_words(only_errors(read_lines("app.log"))):
    print(word)

itertools — მზა იტერატორები

from itertools import count, cycle, islice, chain, product, combinations

# უსასრულო მთვლელი
for n in islice(count(10, 5), 4):
    print(n, end=" ")            # 10 15 20 25

# ციკლური გამეორება
colors = cycle(["წითელი", "მწვანე"])
print(list(islice(colors, 5)))

# სიების შეერთება
print(list(chain([1, 2], [3, 4])))       # [1, 2, 3, 4]

# დეკარტული ნამრავლი
print(list(product("AB", [1, 2])))
# [('A',1), ('A',2), ('B',1), ('B',2)]

# კომბინაციები
print(list(combinations([1, 2, 3], 2)))
# [(1,2), (1,3), (2,3)]
